        super().setup()
        self.camera.background_color = WHITE
        self.header = None
        self.step_strip = None
        self._validate_lesson()

    def _validate_lesson(self):
        for st in (self.EXAM, self.RACE, self.HEIGHT, self.REVERSE, self.CHALLENGE):
            assert 0 < st.k < 100
            assert st.above == 100 - st.k
        assert self.EXAM.above == 30
        assert self.RACE.above == 75
        assert self.HEIGHT.above == 10
        assert self.REVERSE.above == 15
        assert self.CHALLENGE.above == 65
