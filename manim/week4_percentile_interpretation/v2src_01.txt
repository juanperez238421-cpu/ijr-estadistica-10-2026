        "IDENTIFY k",
        "VALUE",
        "BELOW / AT",
        "ABOVE",
        "CONTEXT",
        "CHECK CLAIM",
        "COMMUNICATE",
    )

    EXAM = PercentileStatement(70, 82, "exam score", "points")
    RACE = PercentileStatement(25, 12.4, "100 m time", "s")
    HEIGHT = PercentileStatement(90, 175, "height", "cm")
    REVERSE = PercentileStatement(85, 88, "final test score", "points")
    CHALLENGE = PercentileStatement(35, 14.2, "reaction time", "ms")

    def setup(self):
        super().setup()
        self.camera.background_color = WHITE
        self.header = None
        self.step_strip = None
        self._validate_lesson()

    def _validate_lesson(self):
        for st in (self.EXAM, self.RACE, self.HEIGHT, self.REVERSE, self.CHALLENGE):
            assert 0 < st.k < 100
            assert st.above == 100 - st.k
        assert self.EXAM.above == 30
        assert self.RACE.above == 75
        assert self.HEIGHT.above == 10
        assert self.REVERSE.above == 15
        assert self.CHALLENGE.above == 65

    # ------------------------------------------------------------------
    # Typography / layout helpers
    # ------------------------------------------------------------------
    def t(self, content, size=29, weight=NORMAL, color=INK, **kwargs):
        return Text(
            content,
            font_size=size,
            color=color,
            weight=weight,
            line_spacing=0.92,
            **kwargs,
        )

    def m(self, expr, size=40, color=INK, **kwargs):
        return MathTex(expr, font_size=size, color=color, **kwargs)

    def fit(self, mob, w=SAFE_W, h=5.65):
        if mob.width > w:
            mob.scale_to_fit_width(w)
        if mob.height > h:
            mob.scale_to_fit_height(h)
        return mob

    def panel(self, width, height, fill=PAPER, stroke=LIGHT, sw=1.5):
        return RoundedRectangle(
            width=width,
            height=height,
            corner_radius=0.12,
            stroke_color=stroke,
            stroke_width=sw,
            fill_color=fill,
            fill_opacity=1,
        )

    def note(self, text, width=11.0, height=0.82, size=25, fill=VERY_LIGHT):
        box = self.panel(width, height, fill=fill)
        tx = self.t(text, size, NORMAL, color=DARK)
        self.fit(tx, w=width - 0.45, h=height - 0.18)
        tx.move_to(box)
        return VGroup(box, tx)

    def formula_panel(self, tex, width=8.4, height=1.05, size=43):
        box = self.panel(width, height, fill=WHITE, stroke=DARK, sw=1.6)
        eq = self.m(tex, size)
        self.fit(eq, w=width - 0.55, h=height - 0.22)
        eq.move_to(box)
        return VGroup(box, eq)

    def set_header(self, section, subtitle):
        if self.header is not None and self.header in self.mobjects:
            self.remove(self.header)
        label = self.t(section, 23, BOLD, color=DARK).to_edge(UL, buff=0.42)
        available = 14.85 - label.width - 0.38
        sub = self.t(subtitle, 24, NORMAL, color=MID)
        if sub.width > available:
            sub.scale_to_fit_width(max(available, 4.0))
        sub.next_to(label, RIGHT, buff=0.34)
        line = Line(LEFT * 7.55, RIGHT * 7.55, color=LIGHT, stroke_width=1.6)
        line.next_to(label, DOWN, buff=0.20).align_to(label, LEFT)
        self.header = VGroup(label, sub, line)
        self.add(self.header)

    def clear_stage(self, keep_header=True, keep_steps=False):
        keep = set()
        if keep_header and self.header is not None:
            keep.add(self.header)
        if keep_steps and self.step_strip is not None:
            keep.add(self.step_strip)
