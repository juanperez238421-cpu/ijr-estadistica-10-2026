    def worked_percentile(self, k, title, subtitle, decile_name=None):
        sol = percentile_course(self.DATA_A, k)
        self.set_header(title, subtitle)
        self.set_step(1)
        raw = self.data_row(sol.raw, 1.90, "RAW")
        self.play(FadeIn(raw), run_time=RUN)
        ordered = self.data_row(sol.ordered, 0.92, "ORDERED")
        self.play(TransformFromCopy(raw, ordered), run_time=RUN_SLOW)
        self.wait(PAUSE_READ)

        self.set_step(2)
        nline = self.m(rf"n={len(sol.ordered)}", 40).move_to([0, -0.05, 0])
        self.play(Write(nline), run_time=RUN)

        self.set_step(3)
        if decile_name:
            choose = self.m(rf"{decile_name}=P_{{{int(k)}}}\quad\Rightarrow\quad k={int(k)}", 39).move_to([0, -0.72, 0])
        else:
            choose = self.m(rf"P_{{{int(k)}}}\quad\Rightarrow\quad k={int(k)}", 40).move_to([0, -0.72, 0])
        self.play(Write(choose), run_time=RUN)

        self.set_step(4)
        loc = self.m(rf"L=\frac{{{int(k)}}}{{100}}({len(sol.ordered)}+1)={fmt(sol.locator)}", 41).move_to([0, -1.43, 0])
        self.play(Write(loc), run_time=RUN_SLOW)

        indexed = self.indexed_row(sol.ordered, y=0.75)
        self.play(FadeOut(raw), FadeOut(ordered), FadeOut(nline), FadeOut(choose), FadeOut(loc), run_time=RUN_FAST)
        self.play(FadeIn(indexed), run_time=RUN)

        self.set_step(5)
        cards, indices = indexed
        for pos in sorted(set((sol.lower_position, sol.upper_position))):
            card = cards[pos - 1]
            self.play(Indicate(card, color=DARK, scale_factor=1.08), run_time=RUN_FAST)
        neigh = self.t(
            f"Position {fmt(sol.locator)} uses " +
            (f"x{sol.lower_position} only" if sol.lower_position == sol.upper_position else f"x{sol.lower_position} and x{sol.upper_position}"),
            24,
            BOLD,
            color=DARK,
        ).move_to([0, -0.40, 0])
        self.play(FadeIn(neigh), run_time=RUN)

        self.set_step(6)
        if sol.fraction == 0:
            calc = self.m(rf"P_{{{int(k)}}}=x_{{{sol.lower_position}}}={fmt(sol.value)}", 45).move_to([0, -1.28, 0])
            expl = self.note("L is an integer, so no interpolation is needed.", 9.2, 0.72, 23).move_to([0, -2.06, 0])
        else:
            low = sol.ordered[sol.lower_position - 1]
            high = sol.ordered[sol.upper_position - 1]
            calc = self.m(
                rf"P_{{{int(k)}}}={fmt(low)}+{fmt(sol.fraction)}({fmt(high)}-{fmt(low)})={fmt(sol.value)}",
                42,
            ).move_to([0, -1.28, 0])
            expl = self.note(
                f"L={fmt(sol.locator)} is halfway between positions {sol.lower_position} and {sol.upper_position}.",
                10.8,
                0.72,
                23,
            ).move_to([0, -2.06, 0])
