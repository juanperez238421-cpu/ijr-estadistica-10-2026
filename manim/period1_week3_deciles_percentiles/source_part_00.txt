"""Statistics 10 · Period 1 · Week 3 — Deciles & Percentiles.

Professional theory + worked examples + Percentile Line-Up Game.

Continuity with Week 2:
- Week 2 used the classroom median-of-halves convention for Q1, Q2 and Q3.
- Week 3 generalizes the *idea of position* to deciles and percentiles.
- For percentile-location exercises in this course, this lesson adopts one
  explicit classroom convention:

      L = (k / 100) (n + 1)

  where k is the percentile number and L is a 1-indexed position in the
  ordered sample. If L is not an integer, linear interpolation is used between
  the two neighboring ordered observations.

Important methodological note:
Different textbooks and software packages may implement other sample-quantile
conventions. This lesson states the convention explicitly so students do not
mix algorithms. The Week 2 quartile algorithm is not retroactively replaced.

Eight-step Week 3 recipe:
1. ORDER
2. COUNT n
3. CHOOSE k
4. LOCATE L
5. FIND NEIGHBORS
6. INTERPOLATE
7. STATE RESULT
8. INTERPRET

# ---------------------------------------------------------------------------
# Mathematics / data validation
# ---------------------------------------------------------------------------
@dataclass(frozen=True)
class PercentileSolution:
    raw: tuple[float, ...]
    ordered: tuple[float, ...]
    k: float
    locator: float
    lower_position: int
    upper_position: int
    fraction: float
    value: float


def fmt(v: float) -> str:
    if abs(v - round(v)) < 1e-9:
        return str(int(round(v)))
    return f"{v:.2f}".rstrip("0").rstrip(".")


def percentile_course(values: Sequence[float], k: float) -> PercentileSolution:
    """Course percentile convention L=(k/100)(n+1), with interpolation."""
    if len(values) < 3:
        raise ValueError("At least three observations are required.")
    if not (0 < k < 100):
        raise ValueError("k must satisfy 0 < k < 100.")
    ordered = tuple(sorted(float(v) for v in values))
    n = len(ordered)
    locator = (k / 100.0) * (n + 1)
    # The lesson deliberately chooses tasks with 1 <= L <= n.
    if locator < 1 or locator > n:
        raise ValueError("Chosen percentile falls outside the interior locator range.")
    j = floor(locator)
    fraction = locator - j
    if abs(fraction) < 1e-9:
        lower_position = upper_position = j
        value = ordered[j - 1]
        fraction = 0.0
    else:
        lower_position = j
        upper_position = j + 1
        low = ordered[j - 1]
        high = ordered[j]
        value = low + fraction * (high - low)
    return PercentileSolution(
        raw=tuple(float(v) for v in values),
        ordered=ordered,
        k=float(k),
        locator=float(locator),
        lower_position=lower_position,
        upper_position=upper_position,
        fraction=float(fraction),
        value=float(value),
    )
