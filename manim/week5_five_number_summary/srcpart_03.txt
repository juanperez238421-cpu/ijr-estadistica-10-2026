    def final_recipe(self):
        self.set_header("FINAL RECIPE", "FIVE NUMBERS, ONE CONSISTENT PROCESS")
        left_items = (
            ("1", "ORDER", "Smallest → largest"),
            ("2", "MIN / MAX", "Read endpoints"),
            ("3", "FIND Q2", "Find the median"),
            ("4", "SPLIT", "Lower / upper halves"),
        )
        right_items = (
            ("5", "FIND Q1", "Median of lower half"),
            ("6", "FIND Q3", "Median of upper half"),
            ("7", "WRITE 5-NUM", "(min,Q1,Q2,Q3,max)"),
            ("8", "INTERPRET", "Center + middle spread + endpoints"),
        )

        def recipe_column(items):
            col = VGroup()
            for num, title, desc in items:
                box = self.panel(6.55, 1.02, fill=WHITE)
                n = self.t(num, 24, BOLD, color=DARK)
                tt = self.t(title, 21, BOLD, color=DARK)
                dd = self.t(desc, 18.5, NORMAL, color=MID)
                row = VGroup(n, tt, dd).arrange(RIGHT, buff=0.22)
                self.fit(row, w=6.10, h=0.60)
                row.move_to(box)
                col.add(VGroup(box, row))
            return col.arrange(DOWN, buff=0.15)

        left = recipe_column(left_items)
        right = recipe_column(right_items)
        group = VGroup(left, right).arrange(RIGHT, buff=0.42).move_to([0, 0.20, 0])
        self.play(LaggedStart(*[FadeIn(x) for x in group], lag_ratio=0.07), run_time=RUN_SLOW)
        end = self.note(
            "If the data are not ordered first, every quartile decision after that is at risk.",
            13.1,
            0.86,
            24,
        ).move_to([0, -2.67, 0])
        self.play(FadeIn(end), run_time=RUN)
        self.wait(PAUSE_SUMMARY)
