    def worked_example(self, label, raw, result: FiveNumberResult, odd: bool, focus):
        self.set_header(label, focus)
        self.step_strip = None

        raw_label = self.t("RAW DATA", 20, BOLD, color=MID).move_to([-6.55, 2.38, 0])
        raw_cards = self.data_cards(raw, y=2.18, max_width=12.2, show_indices=False)
        self.play(FadeIn(raw_label), LaggedStart(*[FadeIn(c) for c in raw_cards], lag_ratio=0.06), run_time=RUN_SLOW)
        prompt = self.note("Try it first: build the complete five-number summary.", 10.2, 0.82, 24).move_to([0, 0.82, 0])
        self.play(FadeIn(prompt), run_time=RUN)
        self.wait(PAUSE_WORK)
        self.play(FadeOut(prompt), FadeOut(raw_label), FadeOut(raw_cards), run_time=RUN_FAST)

        # STEP 1 — ORDER
        self.set_step(1)
        ordered_label = self.t("ORDERED DATA", 20, BOLD, color=MID).move_to([-6.30, 2.72, 0])
        ordered = self.data_cards(result.ordered, y=2.26, max_width=12.7, show_indices=True)
        self.play(FadeIn(ordered_label), LaggedStart(*[FadeIn(c) for c in ordered], lag_ratio=0.06), run_time=RUN_SLOW)
        self.wait(PAUSE_READ)

        # STEP 2 — MIN / MAX
        self.set_step(2)
        min_card = self.formula_panel(rf"\min={fmt(result.minimum)}", 3.20, 0.92, 36).move_to([-3.45, 0.90, 0])
        max_card = self.formula_panel(rf"\max={fmt(result.maximum)}", 3.20, 0.92, 36).move_to([3.45, 0.90, 0])
        first_box = ordered[0][0]
        last_box = ordered[-1][0]
        a1 = Arrow(first_box.get_bottom() + DOWN * 0.02, min_card.get_top(), color=DARK, stroke_width=2.0, buff=0.10)
        a2 = Arrow(last_box.get_bottom() + DOWN * 0.02, max_card.get_top(), color=DARK, stroke_width=2.0, buff=0.10)
        self.play(GrowArrow(a1), GrowArrow(a2), FadeIn(min_card), FadeIn(max_card), run_time=RUN_SLOW)
        self.wait(PAUSE_READ)
        self.play(FadeOut(a1), FadeOut(a2), FadeOut(min_card), FadeOut(max_card), run_time=RUN_FAST)

        # STEP 3 — FIND Q2
        self.set_step(3)
        n = len(result.ordered)
        n_note = self.note(f"n = {n} → {'odd' if odd else 'even'} number of observations", 6.7, 0.76, 22).move_to([0, 0.95, 0])
        if odd:
            mid = n // 2
            q2_calc = self.formula_panel(rf"Q_2=x_{{{mid+1}}}={fmt(result.q2)}", 6.0, 0.96, 39).move_to([0, -0.02, 0])
        else:
            left_i = n // 2
            right_i = left_i + 1
            a = result.ordered[left_i - 1]
            b = result.ordered[right_i - 1]
            q2_calc = self.formula_panel(
                rf"Q_2=\frac{{x_{{{left_i}}}+x_{{{right_i}}}}}{{2}}=\frac{{{fmt(a)}+{fmt(b)}}}{{2}}={fmt(result.q2)}",
                8.4,
                0.98,
                38,
            ).move_to([0, -0.02, 0])
        self.play(FadeIn(n_note), FadeIn(q2_calc), run_time=RUN_SLOW)
        self.wait(PAUSE_EXPLAIN)
        self.play(FadeOut(n_note), FadeOut(q2_calc), run_time=RUN_FAST)

        # STEP 4 — SPLIT
        self.set_step(4)
        lower = self.labeled_row("LOWER", result.lower, 0.80, max_width=5.3)
        upper = self.labeled_row("UPPER", result.upper, -0.22, max_width=5.3)
        split_note = self.note(
            "Exclude Q2 before splitting." if odd else "Split evenly: Q2 sits between the two halves.",
            8.6,
            0.72,
            21,
        ).move_to([0, -1.18, 0])
        self.play(FadeIn(lower), FadeIn(upper), run_time=RUN)
        self.play(FadeIn(split_note), run_time=RUN)
        self.wait(PAUSE_EXPLAIN)

        # STEP 5 — FIND Q1
        self.set_step(5)
        la, lb = self._middle_pair(result.lower)
        q1calc = self.quartile_calc("Q_1", la, lb, result.q1, y=-1.92)
        self.play(FadeOut(split_note), FadeIn(q1calc), run_time=RUN)
        self.wait(PAUSE_READ)

        # STEP 6 — FIND Q3
        self.set_step(6)
        ua, ub = self._middle_pair(result.upper)
        q3calc = self.quartile_calc("Q_3", ua, ub, result.q3, y=-1.92)
        self.play(Transform(q1calc, q3calc), run_time=RUN)
        self.wait(PAUSE_READ)

        # STEP 7 — WRITE 5-NUM
        self.set_step(7)
        self.play(FadeOut(lower), FadeOut(upper), FadeOut(q1calc), run_time=RUN_FAST)
        summary = self.five_summary_panel(result, y=-0.65, width=12.8)
        axis, _ = self.five_point_axis(result, y=0.90, width=10.8)
        self.play(Create(axis[0]), run_time=RUN)
        self.play(LaggedStart(*[FadeIn(x) for x in axis[1:]], lag_ratio=0.07), FadeIn(summary), run_time=RUN_SLOW)
        self.wait(PAUSE_EXPLAIN)

        # STEP 8 — INTERPRET
        self.set_step(8)
        interpretation = self.note(
            f"Center: Q2 = {fmt(result.q2)}   ·   Middle 50%: {fmt(result.q1)} to {fmt(result.q3)}   ·   IQR = {fmt(result.iqr)}   ·   Range = {fmt(result.data_range)}",
            13.6,
            0.88,
            22,
        ).move_to([0, -2.28, 0])
        self.play(FadeIn(interpretation), run_time=RUN)
        self.wait(PAUSE_EXPLAIN)
        self.clear_stage(keep_header=False, keep_steps=False)
        self.step_strip = None

    def _middle_pair(self, values):
        n = len(values)
        if n % 2 == 1:
            v = values[n // 2]
            return float(v), float(v)
        return float(values[n // 2 - 1]), float(values[n // 2])
