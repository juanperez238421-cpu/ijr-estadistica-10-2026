"""Statistics 10 · Period 1 · Week 5 — Five-Number Summary.

Senior classroom edition. Direct continuation of Week 4 Percentile Interpretation.

Week 4 emphasized meaning of percentile position in context.
Week 5 consolidates the five structural landmarks of an ordered distribution:

    Minimum, Q1, Q2 (Median), Q3, Maximum

Classroom quartile convention used throughout:
- Always order the data first.
- If n is odd, Q2 is the middle observation and is excluded before forming
  the lower and upper halves.
- If n is even, Q2 is the mean of the two central observations and the data
  split evenly into lower and upper halves.
- Q1 and Q3 are the medians of those halves.

This convention intentionally matches Weeks 1–2. The Week 3 percentile locator
formula is NOT used to recompute quartiles in these small classroom datasets.

Persistent construction route:
1. ORDER
2. MIN / MAX
3. FIND Q2
4. SPLIT
5. FIND Q1
6. FIND Q3
7. WRITE 5-NUM
8. INTERPRET

Target: Manim Community Edition 0.20.x, 1920x1080, 30 fps, white classroom
style. Final render must use literal `-pqh`.
"""

# Data model and validated classroom quartile convention
# ---------------------------------------------------------------------------
@dataclass(frozen=True)
class FiveNumberResult:
    ordered: tuple[float, ...]
    lower: tuple[float, ...]
    upper: tuple[float, ...]
    minimum: float
    q1: float
    q2: float
    q3: float
    maximum: float

    @property
    def five(self) -> tuple[float, float, float, float, float]:
        return (self.minimum, self.q1, self.q2, self.q3, self.maximum)

    @property
    def iqr(self) -> float:
        return self.q3 - self.q1

    @property
    def data_range(self) -> float:
        return self.maximum - self.minimum


def five_number_summary(values: tuple[float, ...]) -> FiveNumberResult:
    ordered = tuple(sorted(values))
    n = len(ordered)
    if n < 5:
        raise ValueError("At least five observations are required for this lesson.")

    if n % 2 == 1:
        middle = n // 2
        q2 = float(ordered[middle])
        lower = ordered[:middle]
        upper = ordered[middle + 1 :]
    else:
        middle = n // 2
        q2 = (ordered[middle - 1] + ordered[middle]) / 2
        lower = ordered[:middle]
        upper = ordered[middle:]

    q1 = float(median(lower))
    q3 = float(median(upper))

    return FiveNumberResult(
        ordered=ordered,
        lower=tuple(lower),
        upper=tuple(upper),
        minimum=float(ordered[0]),
        q1=q1,
        q2=q2,
        q3=q3,
        maximum=float(ordered[-1]),
    )


def fmt(v: float) -> str:
    if abs(v - round(v)) < 1e-9:
        return str(int(round(v)))
    return f"{v:.2f}".rstrip("0").rstrip(".")

        if mob.width > w:

        if mob.height > h:

        box = self.panel(width, height, fill=fill)
        tx = self.t(text, size, NORMAL, color=DARK)
        self.fit(tx, w=width - 0.45, h=height - 0.18)
        tx.move_to(box)
        return VGroup(box, tx)

        eq = self.m(tex, size)
        self.fit(eq, w=width - 0.55, h=height - 0.22)
        eq.move_to(box)
        return VGroup(box, eq)

    def set_header(self, section, subtitle):
        if self.header is not None and self.header in self.mobjects:
            self.remove(self.header)
        label = self.t(section, 23, BOLD, color=DARK).to_edge(UL, buff=0.42)
        available = 14.85 - label.width - 0.38
        sub = self.t(subtitle, 24, NORMAL, color=MID)
        if sub.width > available:
            sub.scale_to_fit_width(max(available, 4.0))
        sub.next_to(label, RIGHT, buff=0.34)
        line = Line(LEFT * 7.55, RIGHT * 7.55, color=LIGHT, stroke_width=1.6)
        line.next_to(label, DOWN, buff=0.20).align_to(label, LEFT)
        self.header = VGroup(label, sub, line)
        self.add(self.header)

    def clear_stage(self, keep_header=True, keep_steps=False):
        keep = set()
        if keep_header and self.header is not None:
            keep.add(self.header)
        if keep_steps and self.step_strip is not None:
            keep.add(self.step_strip)
        removable = [mob for mob in self.mobjects if mob not in keep]
        if removable:
            self.play(*[FadeOut(mob) for mob in removable], run_time=RUN_FAST)
        if keep_header and self.header is not None and self.header not in self.mobjects:
            self.add(self.header)
        if keep_steps and self.step_strip is not None and self.step_strip not in self.mobjects:
            self.add(self.step_strip)
