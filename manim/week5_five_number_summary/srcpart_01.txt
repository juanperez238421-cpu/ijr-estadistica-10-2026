    def data_cards(self, values, y=Y_DATA, max_width=13.4, show_indices=True):
        values = tuple(values)
        n = len(values)
        card_w = min(1.00, (max_width - 0.12 * (n - 1)) / n)
        card_h = 0.72
        cards = VGroup()
        for i, value in enumerate(values, start=1):
            box = RoundedRectangle(
                width=card_w,
                height=card_h,
                corner_radius=0.08,
                stroke_color=DARK,
                stroke_width=1.4,
                fill_color=WHITE,
                fill_opacity=1,
            )
            val = self.t(fmt(value), 24, BOLD, color=DARK)
            val.move_to(box)
            if show_indices:
                idx = self.t(str(i), 13.5, NORMAL, color=MID)
                idx.next_to(box, DOWN, buff=0.07)
                cards.add(VGroup(box, val, idx))
            else:
                cards.add(VGroup(box, val))
        cards.arrange(RIGHT, buff=0.12)
        if cards.width > max_width:
            cards.scale_to_fit_width(max_width)
        cards.move_to([0, y, 0])
        return cards

    def five_point_axis(self, result: FiveNumberResult, y=0.25, width=11.0, labels=True):
        values = result.five
        vmin, vmax = result.minimum, result.maximum
        span = vmax - vmin
        x0 = -width / 2
        x1 = width / 2
        base = Line([x0, y, 0], [x1, y, 0], color=DARK, stroke_width=3.2)

        names = ("MIN", "Q1", "Q2", "Q3", "MAX")
        group = VGroup(base)
        points = []
        for name, value in zip(names, values):
            x = x0 + (value - vmin) / span * width if span else 0
            tick = Line([x, y - 0.19, 0], [x, y + 0.19, 0], color=DARK, stroke_width=2.4)
            dot = Dot([x, y, 0], radius=0.060, color=DARK)
            points.append([x, y, 0])
            group.add(tick, dot)
            if labels:
                nm = self.t(name, 17, BOLD, color=MID).next_to(dot, UP, buff=0.16)
                vv = self.t(fmt(value), 20, BOLD, color=DARK).next_to(dot, DOWN, buff=0.17)
                group.add(nm, vv)
        return group, points

    def five_summary_panel(self, result: FiveNumberResult, y=Y_RESULT, width=12.6):
        expr = (
            rf"\left(\,{fmt(result.minimum)},\ {fmt(result.q1)},\ "
            rf"{fmt(result.q2)},\ {fmt(result.q3)},\ {fmt(result.maximum)}\,\right)"
        )
        box = self.panel(width, 1.03, fill=WHITE, stroke=DARK, sw=1.8)
        lab = self.t("FIVE-NUMBER SUMMARY", 20, BOLD, color=MID)
        eq = self.m(expr, 40)
        content = VGroup(lab, eq).arrange(RIGHT, buff=0.35)
        self.fit(content, w=width - 0.50, h=0.76)
        content.move_to(box)
        group = VGroup(box, content).move_to([0, y, 0])
        return group

    def quartile_calc(self, symbol, a, b, result, y=Y_CALC):
        if abs(a - b) < 1e-9:
            tex = rf"{symbol}={fmt(result)}"
        else:
            tex = rf"{symbol}=\frac{{{fmt(a)}+{fmt(b)}}}{{2}}={fmt(result)}"
        return self.formula_panel(tex, width=6.2, height=0.94, size=38).move_to([0, y, 0])
